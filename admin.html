Regras & Admins
Configure o Firebase para proteger o painel
Aplique as regras abaixo no Firebase Console → Firestore → Rules para garantir que apenas admins autorizados possam escrever na coleção places.

Regras de segurança do Firestore

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Admins: qualquer admin pode ver e gerenciar a lista.
    // (O 1º admin é criado uma vez no Console; os próximos, pelo painel.)
    match /admins/{email} {
      allow read, write: if request.auth != null
        && exists(/databases/$(database)/documents/admins/$(request.auth.token.email));
    }

    // Lugares: leitura pública, escrita só para admins
    match /places/{placeId} {
      allow read: if true;
      allow write: if request.auth != null
                   && exists(/databases/$(database)/documents/admins/$(request.auth.token.email));
    }

    // Categorias: leitura pública, escrita só para admins
    match /categories/{categoryId} {
      allow read: if true;
      allow write: if request.auth != null
                   && exists(/databases/$(database)/documents/admins/$(request.auth.token.email));
    }

    // Dados de cada usuário — só o dono lê e escreve o que é dele.
    // (Salvos, Quero ir hoje, Visitados, Pulados, avaliações que fez,
    //  curtidas que deu e eventos de XP.)
    match /favorites/{userId}/{document=**}       { allow read, write: if request.auth != null && request.auth.uid == userId; }
    match /want_today/{userId}/{document=**}       { allow read, write: if request.auth != null && request.auth.uid == userId; }
    match /been_there/{userId}/{document=**}        { allow read, write: if request.auth != null && request.auth.uid == userId; }
    match /skipped/{userId}/{document=**}           { allow read, write: if request.auth != null && request.auth.uid == userId; }
    match /ratings_by_user/{userId}/{document=**}   { allow read, write: if request.auth != null && request.auth.uid == userId; }
    match /likes_by_user/{userId}/{document=**}     { allow read, write: if request.auth != null && request.auth.uid == userId; }
    match /xp_events/{userId}/{document=**}          { allow read, write: if request.auth != null && request.auth.uid == userId; }

    // Perfil do usuário (foto, badges, privacidade): só o dono
    match /user_profiles/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Avaliações: média é pública; cada voto pertence ao seu dono
    match /ratings/{placeId} {
      allow read: if true;
      allow write: if request.auth != null;
      match /votes/{userId} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Curtidas: registro pertence ao seu dono
    match /likes/{placeId}/users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // Salvos (espelho público por lugar) — métricas do admin
    match /saves/{placeId}/users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // Já fui (espelho público por lugar) — métricas do admin
    match /visits/{placeId}/users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // Comentários
    match /comments/{document=**} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null
                            && request.auth.uid == resource.data.uid;
    }
  }
}
